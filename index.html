-- ============================================================ -- MÓDULO FINANCEIRO — Connect VIP -- Execute este SQL no editor do Supabase (SQL Editor) -- ============================================================ -- ---------------------------------------------------------------- -- TABELA: boletos -- Armazena cada parcela/boleto de um aluno. -- ---------------------------------------------------------------- CREATE TABLE IF NOT EXISTS boletos ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), student_id UUID NOT NULL REFERENCES students(id) ON DELETE CASCADE, mes_referencia INT NOT NULL CHECK (mes_referencia BETWEEN 1 AND 12), ano_referencia INT NOT NULL DEFAULT EXTRACT(YEAR FROM NOW()), data_vencimento DATE NOT NULL, valor NUMERIC(10,2) NOT NULL CHECK (valor > 0), status TEXT NOT NULL DEFAULT 'aberto' CHECK (status IN ('aberto','pago')), data_pagamento DATE, -- preenchida ao dar baixa observacao TEXT, created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); -- Índices úteis para consultas frequentes CREATE INDEX IF NOT EXISTS idx_boletos_student ON boletos(student_id); CREATE INDEX IF NOT EXISTS idx_boletos_status ON boletos(status); CREATE INDEX IF NOT EXISTS idx_boletos_venc ON boletos(data_vencimento); CREATE INDEX IF NOT EXISTS idx_boletos_ano ON boletos(ano_referencia); -- ---------------------------------------------------------------- -- FUNÇÃO: atualiza updated_at automaticamente -- ---------------------------------------------------------------- CREATE OR REPLACE FUNCTION set_updated_at() RETURNS TRIGGER LANGUAGE plpgsql AS $$ BEGIN NEW.updated_at = NOW(); RETURN NEW; END; $$; CREATE TRIGGER boletos_updated_at BEFORE UPDATE ON boletos FOR EACH ROW EXECUTE FUNCTION set_updated_at(); -- ---------------------------------------------------------------- -- VIEW: resumo_financeiro_aluno -- Facilita a leitura do painel de inadimplência e faturamento. -- ---------------------------------------------------------------- CREATE OR REPLACE VIEW resumo_financeiro_aluno AS SELECT s.id AS student_id, s.first_name || ' ' || s.last_name AS nome_aluno, s.status AS aluno_ativo, COUNT(b.id) AS total_boletos, COUNT(b.id) FILTER (WHERE b.status = 'pago') AS total_pagos, COUNT(b.id) FILTER (WHERE b.status = 'aberto') AS total_abertos, COALESCE(SUM(b.valor) FILTER (WHERE b.status = 'pago'), 0) AS valor_pago, COALESCE(SUM(b.valor) FILTER (WHERE b.status = 'aberto'), 0) AS valor_aberto, COALESCE(SUM(b.valor), 0) AS valor_total FROM students s LEFT JOIN boletos b ON b.student_id = s.id GROUP BY s.id, s.first_name, s.last_name, s.status;
